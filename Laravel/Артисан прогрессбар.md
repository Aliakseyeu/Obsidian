```php
$progressbar = $this->output->createProgressBar($count); // Создание

$progressbar->finish(); // ФИниш
```

```php
// Передаем саму коллекцию $providers первым аргументом 
$this->withProgressBar(
	$providers, 
	static function ($provider) use ($service): void { 
		// Этот код Laravel сам вызовет в цикле для каждого провайдера 
		// и автоматически сдвинет ползунок прогрессбара на +1 
		$service->send($provider); 
	}
);
```
